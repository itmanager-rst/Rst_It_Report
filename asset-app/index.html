'••••' + cfg.token.slice(-4) : ''; res.targets = cfg.targets; res.hasAdminKey = true; res.lastSource = pr.getProperty('LINE_LAST_SOURCE') || ''; } return out(res, cb); } if (action === 'lineSetConfig') { if (role !== 'admin') return out({ status: 'error', code: 'forbidden', msg: 'เฉพาะ IT Admin เท่านั้น' }, cb); if (p.token) pr.setProperty('LINE_TOKEN', String(p.token).trim()); if (p.targets !== undefined) pr.setProperty('LINE_TARGETS', String(p.targets)); if (p.events) { try { pr.setProperty('LINE_EVENTS', JSON.stringify(JSON.parse(p.events))); } catch (err) {} } return out({ status: 'ok', msg: 'saved' }, cb); } if (action === 'lineNotify') { if (!cfg.token || !cfg.targets.length) return out({ status: 'error', msg: 'not_configured' }, cb); var ev = String(p.event || ''); if (ev && cfg.events[ev] === false) return out({ status: 'skipped', msg: 'ปิดแจ้งเตือนประเภท ' + ev }, cb); if (!p.msg) return out({ status: 'error', msg: 'empty message' }, cb); // กันส่งรัว: ไม่เกิน 30 ข้อความ / 10 นาที var c = CacheService.getScriptCache(), n = parseInt(c.get('LINE_RATE') || '0', 10); if (n >= 30) return out({ status: 'error', msg: 'ส่งแจ้งเตือนถี่เกินไป ลองใหม่ภายหลัง' }, cb); c.put('LINE_RATE', String(n + 1), 600); var text = String(p.msg).slice(0, 4400) + '\n\n🔗 เข้าสู่ระบบเพื่อตรวจสอบ/อนุมัติ:\n' + APP_URL; return out(linePushAll(cfg.token, cfg.targets, text), cb); } return out({ status: 'error', msg: 'unknown line action: ' + action }, cb); } // ── Webhook จาก LINE: ตอบรับ HTTP 200 ทุกกรณี ── function lineWebhook(e) { var cfg = lineGetConfig(); var body = {}; try { body = JSON.parse(e.postData.contents); } catch (err) {} if (body.events && Array.isArray(body.events)) { body.events.forEach(function (ev) { var src = ev.source || {}; var id = src.groupId || src.roomId || src.userId; if (!id) return; lineProps().setProperty('LINE_LAST_SOURCE', id + '|' + (src.type || '') + '|' + new Date().toISOString()); var txt = ev.message && ev.message.type === 'text' ? String(ev.message.text).trim().toLowerCase() : ''; if (ev.type === 'join' || ev.type === 'follow' || txt === 'id' || txt === 'ไอดี') { lineReply(cfg.token, ev.replyToken, 'ID สำหรับรับแจ้งเตือน IT Asset:\n' + id + '\n\nนำไปใส่ในหน้า ตั้งค่า → แจ้งเตือน LINE'); } }); } return ContentService.createTextOutput('OK'); }
